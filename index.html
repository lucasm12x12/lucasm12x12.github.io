02 / Engineering

Code Example.

A compact, illustrative Luau example showing strict typing, documented methods, service lifecycle, and a self-based API.

DirectionalMovementServiceClient.luau · illustrative sample
--!strict
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")

local DirectionalMovementServiceClient = {}

type Service = typeof(DirectionalMovementServiceClient) & {
    player: Player,
    humanoid: Humanoid?,
    rootPart: BasePart?,
    currentAngle: number,
    movementConnection: RBXScriptConnection?,
}

--[=[
    Smoothly updates a movement heading relative to the character.

    @param self Service -- The active client service.
    @param dt number -- Elapsed simulation time in seconds.
    @return ()
]=]
function DirectionalMovementServiceClient.Update(
    self: Service,
    dt: number
): ()
    local humanoid = self.humanoid
    local rootPart = self.rootPart
    if not humanoid or not rootPart then
        return
    end

    local localDirection: Vector3 =
        rootPart.CFrame:VectorToObjectSpace(humanoid.MoveDirection)

    local targetAngle: number = 0
    if localDirection.Magnitude > 0.01 then
        targetAngle = math.atan2(localDirection.X, -localDirection.Z)
    end

    local difference = math.atan2(
        math.sin(targetAngle - self.currentAngle),
        math.cos(targetAngle - self.currentAngle)
    )
    local alpha = 1 - math.exp(-12 * dt)
    self.currentAngle += difference * alpha
end

--[=[
    Binds the character and starts the simulation loop.

    @param self Service -- The active client service.
    @return ()
]=]
function DirectionalMovementServiceClient.Init(self: Service): ()
    self.player = Players.LocalPlayer
    self.currentAngle = 0
    self.humanoid = nil
    self.rootPart = nil

    local function bindCharacter(character: Model): ()
        self.humanoid = character:WaitForChild("Humanoid") :: Humanoid
        self.rootPart = character:WaitForChild("HumanoidRootPart") :: BasePart
        self.currentAngle = 0
    end

    self.player.CharacterAdded:Connect(bindCharacter)
    if self.player.Character then
        bindCharacter(self.player.Character)
    end

    self.movementConnection = RunService.PreSimulation:Connect(
        function(dt: number)
            self:Update(dt)
        end
    )
end

return DirectionalMovementServiceClient :: Service

This sample is written to demonstrate coding conventions; it is not a direct copy of a production module. Replace it with a real, reviewed excerpt when you're ready.
